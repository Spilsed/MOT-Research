	\section{Introduction to Topology}
		\subsection{Topological Spaces and Axioms}
		A topological space is an ordered pair $(X,\tau)$, where $X$ is a set and $\tau$ a topology on $X$, or a set of subsets of $X$ (these are open sets) satisfying the following axioms:
		\begin{enumerate}
			\item The empty set and the entire set ($X$) are in $\tau$.
			\item The union of any collection of sets in $\tau$ is also in $\tau$: $$\bigcup_{\alpha\in I}U_{\alpha}\in \tau\text{ for all }U_{\alpha}\in\tau$$
			\item The intersection of any finite collection of sets in $\tau$ are in $\tau$: $$\bigcup_{i=1}^n U_i\in t\text{ for all }U_1,...,U_n\in t$$
		\end{enumerate}
		\subsection{Open Sets and Closed Sets}
		\begin{itemize}
			\item \textbf{Open Sets:} The elements of a topology $\tau$ are called open sets. They are the foundational blocks for defining continuity, convergence, and limit points.
			\item \textbf{Closed Sets:} A subset $F\subseteq X$ is called a closed set if its complement $X\setminus F$ is an open set. Arbitrary intersections of closed sets are closed, and finite unions of closed sets are closed (by De Morgan's laws).
		\end{itemize}
		\subsection{Neighborhoods and Bases}
		\begin{itemize}
			\item \textbf{Neighborhoods:} Let $x \in X$. A subset $N\subseteq X$ is called a neighborhood of $x$ if there exists an open set $U\in \tau$ such that $x\in U\subseteq N$.
			\item \textbf{Base:} A sub-collection $\mathcal{B}\subseteq\tau$ is a base/basis for the topology $\tau$ if every open set in $\tau$ can be expressed as a union of elements from $\mathcal{B}$.
			\item \textbf{Subbase:} A collection $\mathcal{S}$ of subsets of $X$ is a subbase if the collection of all the finite intersections of elements of $\mathcal{S}$ forms a base for $\tau$.
		\end{itemize}
